/**

 * @property {number} yylineno - Número de línea actual
 * @property {LexerLocation} yylloc - Ubicación actual del token
 * @property {string} yytext - Texto del token acumulado
 * @property {string} match - Coincidencia actual
 * @property {Array} matches - Grupos de captura del regex
 * @property {string} matched - Todo el texto coincidido hasta ahora

/**

 * @property {boolean} [flex] - Usa modo FLEX (longest match en lugar de first match)
 * @property {boolean} [ranges] - Habilita información de rango de bytes
 * @property {boolean} [case-insensitive] - Las reglas regex serán case-insensitive

var lexer = (function(){
var lexer = ({

EOF: 1,

/**
 * Maneja errores de análisis léxico.
 * 
 * @internal
 * @param {string} str - Mensaje de error
 * @param {Object} hash - Información adicional sobre el error
 * @throws {Error} Si no hay parser configurado
 */
parseError: function parseError(str, hash) {
    if (this.yy.parser) {
        this.yy.parser.parseError(str, hash);
    } else {
        throw new Error(str);
    }
},

/**
 * Reinicia el lexer y configura una nueva entrada para analizar.
 * 
 * Este método limpia todo el estado del lexer y lo prepara para procesar
 * un nuevo flujo de entrada. Debe llamarse antes de usar lex() por primera vez
 * con una entrada diferente.
 * 
 * @internal
 * @param {string} input - Cadena de entrada a analizar
 * @param {Object} [yy] - Contexto compartido del parser (opcional)
 * @param {Object} [yy.parser] - Referencia al parser para manejar errores
 * @returns {Object} Retorna this para permitir encadenamiento de métodos
 * 
 * @example
 * lexer.setInput("hello world");
 * // o con contexto del parser
 * lexer.setInput(input, parserContext);
 */
/**
 * Reinicia el lexer y configura una nueva entrada.
 * @internal
 * @param {string} input - Cadena de entrada a analizar
 * @param {Object} [yy] - Contexto compartido del parser
 * @returns {Object} Retorna this para encadenamiento de métodos
 */

setInput: function(input, yy) {
    this.yy = yy || this.yy || {};
    this._input = input;
    this._more = this._backtrack = this.done = false;
    this.yylineno = this.yyleng = 0;
    this.yytext = this.matched = this.match = '';
    this.conditionStack = ['INITIAL'];
    this.yylloc = {
        first_line: 1,
        first_column: 0,
        last_line: 1,
        last_column: 0
    };
    if (this.options.ranges) {
        this.yylloc.range = [0,0];
    }
    this.offset = 0;
    return this;
},

/**
 * Consume y retorna un carácter de la entrada.
 * 
 * Este método es un componente bajo nivel que procesa un carácter a la vez.
 * Actualiza automáticamente:
 * - Contador de líneas y columnas (yylineno, yylloc)
 * - Texto acumulado (yytext, yyleng)
 * - Posición en la entrada
 * 
 * @internal
 * @returns {string} El carácter consumido
 * 
 * @example
 * // Uso interno en acciones de reglas
 * while (someCondition) {
 *     var char = lexer.input();
 *     if (char === '\n') {
 *         // Manejar nueva línea
 *     }
 * }
 */
input: function () {
    var ch = this._input[0];
    this.yytext += ch;
    this.yyleng++;
    this.offset++;
    this.match += ch;
    this.matched += ch;
    var lines = ch.match(/(?:\r\n?|\n).*/g);
    if (lines) {
        this.yylineno++;
        this.yylloc.last_line++;
    } else {
        this.yylloc.last_column++;
    }
    if (this.options.ranges) {
        this.yylloc.range[1]++;
    }

    this._input = this._input.slice(1);
    return ch;
},

/**
 * Retrocede caracteres en la entrada.
 * 
 * Deshace el consumo de caracteres previamente procesados por input().
 * Útil cuando una regla necesita "devolver" caracteres porque pertenecen
 * al siguiente token.
 * 
 * @internal
 * @param {string} ch - Carácter(es) a retroceder
 * @returns {Object} Retorna this
 * 
 * @example
 * // En una acción: si consumimos un carácter que no debería ser parte del token
 * this.input();  // Consumir un carácter
 * this.unput(ch);  // Devolverlo para la siguiente regla
 */
/**
 * Retrocede caracteres en la entrada.
 * @internal
 * @param {string} ch - Carácter(es) a retroceder
 * @returns {Object} Retorna this
 */

unput: function(ch) {
    var len = ch.length;
    var lines = ch.split(/(?:\r\n?|\n)/g);

    this._input = ch + this._input;
    this.yytext = this.yytext.substr(0, this.yytext.length - len);
    this.offset -= len;
    var oldLines = this.match.split(/(?:\r\n?|\n)/g);
    this.match = this.match.substr(0, this.match.length - 1);
    this.matched = this.matched.substr(0, this.matched.length - 1);

    if (lines.length - 1) {
        this.yylineno -= lines.length - 1;
    }
    var r = this.yylloc.range;

    this.yylloc = {
        first_line: this.yylloc.first_line,
        last_line: this.yylineno + 1,
        first_column: this.yylloc.first_column,
        last_column: lines ?
            (lines.length === oldLines.length ? this.yylloc.first_column : 0)
             + oldLines[oldLines.length - lines.length].length - lines[0].length :
          this.yylloc.first_column - len
    };

    if (this.options.ranges) {
        this.yylloc.range = [r[0], r[0] + this.yyleng - len];
    }
    this.yyleng = this.yytext.length;
    return this;
},

/**
 * Señala al lexer que debe continuar acumulando texto en el token actual.
 * 
 * Cuando se llama en una acción de regla, el siguiente token coincidente
 * será anexado al token actual en lugar de ser tratado como un token separado.
 * Esto es útil para tokens multi-línea o secuencias de caracteres que se
 * analizan en fases.
 * 
 * @internal
 * @returns {Object} Retorna this
 * 
 * @example
 * // Regla que continúa en la siguiente coincidencia
 * [{
 *     regex: /part1/,
 *     action: function() { 
 *         this.more();  // Señala que seguirá más
 *         return;  // No retorna token aún
 *     }
 * }]
 * 
 * @see reject
 */
more: function () {
    this._more = true;
    return this;
},

/**
 * Rechaza el token actual e intenta la siguiente regla de lexing.
 * 
 * Usado en lexers con backtracking. Cuando una acción llama reject(),
 * el lexer restaura su estado previo a esta coincidencia y prueba la
 * siguiente regla.
 * 
 * IMPORTANTE: Solo funciona si options.backtrack_lexer está habilitado.
 * 
 * @internal
 * @returns {false|Error} Retorna false si backtracking está habilitado,
 *                        lanza Error si no está habilitado
 * 
 * @throws {Error} Si se llama reject() sin backtrack_lexer habilitado
 * 
 * @example
 * // En modo backtrack
 * [{
 *     regex: /pattern1|pattern2/,
 *     action: function() {
 *         if (someCondition) {
 *             this.reject();  // Deshace esta coincidencia
 *         }
 *         return 'TOKEN';
 *     }
 * }]
 * 
 * @see more
 */
reject: function () {
    if (this.options.backtrack_lexer) {
        this._backtrack = true;
    } else {
        return this.parseError('Lexical error on line ' + (this.yylineno + 1) + '. ' +
            'You can only invoke reject() in the lexer when the lexer is of the ' +
            'backtracking persuasion (options.backtrack_lexer = true).\n' + 
            this.showPosition(), {
            text: "",
            token: null,
            line: this.yylineno
        });
    }
    return this;
},

/**
 * Retiene solo los primeros n caracteres de la coincidencia actual.
 * 
 * Los caracteres restantes son retrocedidos en la entrada para ser
 * procesados como parte del siguiente token.
 * 
 * @internal
 * @param {number} n - Número de caracteres a retener
 * @returns {undefined}
 * 
 * @example
 * // Regla que solo quiere los primeros 2 caracteres de una coincidencia de 4
 * this.less(2);  // Mantiene 2 chars, retrocede 2
 */
/**
 * Retiene solo los primeros n caracteres de la coincidencia actual.
 * @internal
 * @param {number} n - Número de caracteres a retener
 */

less: function(n) {
    this.unput(this.match.slice(n));
},

/**
 * Retorna el texto ya procesado y consumido.
 * 
 * Útil para debugging y mensajes de error. Muestra lo que el lexer ha
 * procesado hasta ahora (últimos 20 caracteres).
 * 
 * @internal
 * @returns {string} Texto consumido hasta ahora (máximo 20 caracteres)
 */
pastInput: function () {
    var past = this.matched.substr(0, this.matched.length - this.match.length);
    return (past.length > 20 ? '...':'') + past.substr(-20).replace(/\n/g, "");
},

/**
 * Retorna el texto próximo a ser procesado.
 * 
 * Útil para debugging y mensajes de error. Muestra lo que sigue en la entrada.
 * 
 * @internal
 * @returns {string} Texto pendiente (máximo 20 caracteres)
 */
upcomingInput: function () {
    var next = this.match;
    if (next.length < 20) {
        next += this._input.substr(0, 20-next.length);
    }
    return (next.substr(0,20) + (next.length > 20 ? '...' : '')).replace(/\n/g, "");
},

/**
 * Muestra la posición exacta del error de análisis léxico.
 * 
 * Formato:
 * ```
 * [texto previo][próximo texto]
 * ___________________^
 * ```
 * 
 * @internal
 * @returns {string} Representación visual del error
 */
showPosition: function () {
    var pre = this.pastInput();
    var c = new Array(pre.length + 1).join("-");
    return pre + this.upcomingInput() + "\n" + c + "^";
},

/**
 * Evalúa si un patrón coincide con la entrada actual y procesa el token.
 * 
 * Este es el método nuclear del análisis léxico. Cuando el lexer encuentra
 * una coincidencia de patrón regex, test_match():
 * 1. Actualiza las posiciones de línea y columna
 * 2. Acumula el texto coincidente (yytext)
 * 3. Llama a performAction para ejecutar el código de la regla
 * 4. En modo backtrack, restaura el estado si la acción rechaza el match
 * 
 * El valor retornado determina:
 * - Token válido (string o número) → se retorna al parser
 * - false → la regla rechazó el match, intenta la siguiente
 * - undefined/null → el token fue consumido sin valor (ej: espacios)
 * 
 * @internal
 * @param {Array} match - Resultado del match del regex
 *                       match[0] = texto completo coincidente
 *                       match[1...n] = grupos de captura opcionales
 * @param {number} indexed_rule - Índice de la regla en el array rules[]
 * @returns {string|number|false} 
 *          - string o number: token a retornar al parser
 *          - false: coincidencia rechazada (backtrack), intenta siguiente regla
 *          - undefined/null/falsy: input consumido sin token
 * 
 * @throws {Error} Si se llama reject() fuera de modo backtrack_lexer
 * 
 * @example
 * // Uso interno en el método next()
 * const match = this._input.match(this.rules[0]);
 * if (match) {
 *     const token = this.test_match(match, 0);
 *     if (token !== false) {
 *         return token;  // Token válido
 *     }
 *     // Si es false, intenta siguiente regla
 * }
 * 
 * @see next
 * @see performAction
 */
/**
 * Updates the lexer state to reflect the match found, and then calls the lexer action function.
 * 
 * @internal                      // Marca como no-pública
 * @param {Array} match           // [fullMatch, group1, ...]
 * @param {number} indexed_rule   // Índice de la regla
 * @returns {string|number|false} // Token o false si rechazado
 * @see next, performAction, reject
 */

test_match: function(match, indexed_rule) {
    var token,
        lines,
        backup;

    if (this.options.backtrack_lexer) {
        // Guardar estado completo para recuperación si reject() es llamado
        backup = {
            yylineno: this.yylineno,
            yylloc: {
                first_line: this.yylloc.first_line,
                last_line: this.last_line,
                first_column: this.yylloc.first_column,
                last_column: this.yylloc.last_column
            },
            yytext: this.yytext,
            match: this.match,
            matches: this.matches,
            matched: this.matched,
            yyleng: this.yyleng,
            offset: this.offset,
            _more: this._more,
            _input: this._input,
            yy: this.yy,
            conditionStack: this.conditionStack.slice(0),
            done: this.done
        };
        if (this.options.ranges) {
            backup.yylloc.range = this.yylloc.range.slice(0);
        }
    }

    // Actualizar líneas según saltos de línea en la coincidencia
    lines = match[0].match(/(?:\r\n?|\n).*/g);
    if (lines) {
        this.yylineno += lines.length;
    }
    
    // Actualizar ubicación del token (línea, columna)
    this.yylloc = {
        first_line: this.yylloc.last_line,
        last_line: this.yylineno + 1,
        first_column: this.yylloc.last_column,
        last_column: lines ?
                     lines[lines.length - 1].length - lines[lines.length - 1].match(/\r?\n?/)[0].length :
                     this.yylloc.last_column + match[0].length
    };
    
    // Acumular el texto coincidente
    this.yytext += match[0];
    this.match += match[0];
    this.matches = match;
    this.yyleng = this.yytext.length;
    
    if (this.options.ranges) {
        this.yylloc.range = [this.offset, this.offset += this.yyleng];
    }
    
    this._more = false;
    this._backtrack = false;
    this._input = this._input.slice(match[0].length);
    this.matched += match[0];
    
    // Ejecutar la acción de la regla
    token = this.performAction.call(this, this.yy, this, indexed_rule, 
                                     this.conditionStack[this.conditionStack.length - 1]);
    
    if (this.done && this._input) {
        this.done = false;
    }
    
    if (token) {
        return token;
    } else if (this._backtrack) {
        // La acción llamó reject() - restaurar estado y retornar false
        for (var k in backup) {
            this[k] = backup[k];
        }
        return false;
    }
    
    return false;
},

/**
 * Retorna el siguiente token del flujo de entrada.
 * 
 * Este es el método principal del lexer. Coordina:
 * 1. Obtención de las reglas actuales según condición de inicio
 * 2. Matching del patrón contra la entrada
 * 3. Ejecución de test_match para procesar coincidencias
 * 4. Manejo de backtracking si está habilitado
 * 5. Retorno del token o EOF
 * 
 * @internal
 * @returns {string|number|boolean} 
 *          - Token coincidente (string o número)
 *          - this.EOF (valor 1) si se agotó la entrada
 *          - false si consumió input sin generar token (ej: comentarios)
 * 
 * @example
 * lexer.setInput("123 + 456");
 * const token1 = lexer.next();  // Retorna NUMBER (o similar)
 * const token2 = lexer.next();  // Retorna PLUS
 * const token3 = lexer.next();  // Retorna NUMBER
 * const eof = lexer.next();     // Retorna 1 (EOF)
 */
next: function () {
    if (this.done) {
        return this.EOF;
    }
    if (!this._input) {
        this.done = true;
    }

    var token,
        match,
        tempMatch,
        index;
    if (!this._more) {
        this.yytext = '';
        this.match = '';
    }
    
    var rules = this._currentRules();
    for (var i = 0; i < rules.length; i++) {
        tempMatch = this._input.match(this.rules[rules[i]]);
        if (tempMatch && (!match || tempMatch[0].length > match[0].length)) {
            match = tempMatch;
            index = i;
            if (this.options.backtrack_lexer) {
                token = this.test_match(tempMatch, rules[i]);
                if (token !== false) {
                    return token;
                } else if (this._backtrack) {
                    match = false;
                    continue;
                } else {
                    return false;
                }
            } else if (!this.options.flex) {
                break;
            }
        }
    }
    
    if (match) {
        token = this.test_match(match, rules[index]);
        if (token !== false) {
            return token;
        }
        return false;
    }
    
    if (this._input === "") {
        return this.EOF;
    } else {
        return this.parseError('Unexpected character: ' + this.showPosition(), {
            text: this._input,
            token: null,
            line: this.yylineno
        });
    }
},
